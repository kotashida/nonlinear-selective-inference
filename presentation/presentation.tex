\begin{frame}{Unadjusted vs. Random：実験設定}
  \textbf{グローバル帰無仮説}
  \[
    \begin{gathered}
      X_{ij}\overset{\mathrm{i.i.d.}}{\sim}\mathcal N(0,1),\qquad
      y_i\overset{\mathrm{i.i.d.}}{\sim}\mathcal N(0,1),\qquad
      X\perp y,\qquad \sigma^2=1\ \text{（既知）}.
    \end{gathered}
  \]

  \vspace{0.15cm}
  \begin{itemize}
    \item $n=100$, $d=20$, 選択数 $k=1$
    \item 反復数 $R=1000$, 有意水準 $\alpha=0.05$
    \item Random Forest：50 trees, max depth 5
    \item \textbf{Random：}$y$ と独立に1特徴を選択
    \item \textbf{Unadjusted SHAP：}平均絶対 SHAP 値が最大の特徴を選択
  \end{itemize}
\end{frame}

\begin{frame}{既知分散：カイ検定}
  選択特徴 $j$ の B-spline 効果を表す正規直交基底を
  $Q_j\in\mathbb R^{n\times r}$（$r=3$）とする。

  \vspace{0.3cm}
  \[
    T_j
    =\frac{\lVert Q_j^\top y\rVert_2}{\sigma}
    =\lVert Q_j^\top y\rVert_2.
  \]

  $H_0$ の下で，特徴 $j$ があらかじめ固定されていれば
  \[
    Q_j^\top y\sim\mathcal N_r(0,I_r)
    \quad\Longrightarrow\quad
    T_j\sim\chi_r=\chi_3,
  \]
  したがって $p_j=P(\chi_3\ge T_j^{\mathrm{obs}})$ と計算する。
\end{frame}

\begin{frame}{SHAP 選択領域の構成}
  \begin{enumerate}
    \item \textbf{応答を1次元に制限する}
      \[
        \begin{gathered}
          P_j=Q_jQ_j^\top,\qquad
          a_j=(I-P_j)y,\qquad
          u_j=\frac{P_jy}{\lVert P_jy\rVert_2},\\
          y(z)=a_j+u_j z,\qquad z\ge 0
          \quad(\sigma=1).
        \end{gathered}
      \]

    \item \textbf{特徴 $j$ が再び選ばれる範囲を求める}
      \[
        \mathcal Z_j
        =\left\{z\ge0:
        j\in\widehat S_{\mathrm{SHAP}}\bigl(X,y(z)\bigr)\right\}.
      \]

    \item \textbf{選択を条件づける}

      $H_0$ 下の $z\sim\chi_r$ を，$\mathcal Z_j$ 上で再正規化する。
  \end{enumerate}
\end{frame}
